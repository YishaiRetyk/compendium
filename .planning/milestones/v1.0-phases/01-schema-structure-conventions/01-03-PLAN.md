    ```python
    python3 -c "
    import yaml, sys
    for f in ['wiki/index.md', 'wiki/log.md']:
        content = open(f).read()
        # Extract frontmatter between --- delimiters
        parts = content.split('---', 2)
        if len(parts) < 3:
            print(f'FAIL: {f} has no valid frontmatter delimiters')
            sys.exit(1)
        try:
            fm = yaml.safe_load(parts[1])
        except yaml.YAMLError as e:
            print(f'FAIL: {f} frontmatter YAML parse error: {e}')
            sys.exit(1)
        # Check required fields exist
        required = ['id', 'title', 'type', 'status', 'summary', 'created_at', 'updated_at', 'sources', 'epistemic_status', 'tags', 'domains', 'privacy']
        missing = [k for k in required if k not in fm]
        if missing:
            print(f'FAIL: {f} missing frontmatter fields: {missing}')
            sys.exit(1)
        # Check field value validity
        if fm['type'] not in ['entity', 'concept', 'source', 'comparison', 'overview']:
            print(f'FAIL: {f} invalid type: {fm[\"type\"]}')
            sys.exit(1)
        if fm['status'] not in ['active', 'stale', 'superseded', 'archived']:
            print(f'FAIL: {f} invalid status: {fm[\"status\"]}')
            sys.exit(1)
        if fm['epistemic_status'] not in ['sourced', 'mixed', 'tentative', 'stale']:
            print(f'FAIL: {f} invalid epistemic_status: {fm[\"epistemic_status\"]}')
            sys.exit(1)
        if fm['privacy'] not in ['local_only', 'cloud_safe']:
            print(f'FAIL: {f} invalid privacy: {fm[\"privacy\"]}')
            sys.exit(1)
        print(f'PASS: {f} frontmatter valid')
    print('All YAML validation passed')
    "
    ```
