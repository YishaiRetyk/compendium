**normalize_link() — exact implementation (D-02)**:
```python
_PLURAL_MAP = {
    'contexts': 'context',
    'policies': 'policy',
    'contracts': 'contract',
}
_PUNCT_RE = re.compile(r'[^\w\s]')   # strips parens, hyphens, punctuation characters
_WS_RE    = re.compile(r'\s+')

def normalize_link(text):
    """D-02: casefold, strip punctuation chars (parens stripped as CHARACTERS not
    content -- 'Hack (Agentive Stack)' -> 'hack agentive stack', NOT 'hack'),
    collapse whitespace, apply small explicit plural map.
    NO edit distance, NO stemmer, NO substring matching."""
    s = text.casefold()
    s = _PUNCT_RE.sub(' ', s)   # parens/hyphen/punct -> space (NOT their content)
    s = s.replace('_', ' ')     # re \w includes underscore; handle separately
    s = _WS_RE.sub(' ', s).strip()
    words = [_PLURAL_MAP.get(w, w) for w in s.split()]
    return ' '.join(words)
```

**_apply_self_alias_fix() helper — exact implementation (CONFIRMED HIGH BUG #2 fix)**:

VERIFIED LIVE (orchestrator): two real wiki pages have colon-space titles that the OLD unquoted emit (`'\n'.join(f'  - {a}' for a in new_aliases)`) would corrupt — YAML parses `  - SC1 Reframing: examples/ Forms...` as a DICT, not a string, breaking idempotency (next run still sees it unreachable → re-fires → compounding corruption):
  - `wiki/decisions/dr-2026-05-01-complementary-systems-boundary.md` — title `Complementary Systems Boundary: Compendium as Durable Wiki Memory in a Multi-System Stack`
  - `wiki/decisions/dr-2026-06-02-sc1-examples-isolable-subgraph.md` — title `SC1 Reframing: examples/ Forms a Visually Isolable Sub-Graph`

**Fixes applied below:** (a) every emitted alias value is YAML-double-quoted with `"` and `\` escaped — safe for titles containing `:`, `#`, `[`, `]`, leading `*`/`&`/`?`/`!`; (b) the helper is called ONCE per page with BOTH candidate values (title + id) passed together — no stale-`fm` second call (review F / Codex); (c) `except` emits a stderr WARNING instead of silently swallowing (so batch-fix failures during Plan 03's 45-page run are visible).
```python
def _yaml_quote_alias(a):
    """YAML-double-quote a scalar so colons/brackets/leading-indicators are safe."""
    s = str(a).replace('\\', '\\\\').replace('"', '\\"')
    return f'"{s}"'

def _apply_self_alias_fix(fpath, fm, to_add):
    """Idempotent: append the already-computed missing aliases (to_add, a list of
    raw string values) to the page's aliases block. Caller computes to_add ONCE per
    page (title and/or id) and calls this ONCE — no stale-fm second call. Never
    removes existing aliases. Emits YAML-double-quoted values so colon-bearing titles
    (e.g. 'Topic: Subtitle') stay STRINGS, not dicts (CONFIRMED HIGH BUG #2)."""
    if not to_add:
        return  # nothing to do — idempotent no-op
    try:
        content = open(fpath, encoding='utf-8').read()
        # Frontmatter section only (up to second '---') — avoid body 'aliases:' blocks (Pitfall 2)
        fm_end = content.find('\n---', 3)  # skip opening ---
        fm_section = content[:fm_end] if fm_end != -1 else content
        existing = [str(a) for a in (fm.get('aliases') or []) if a]
        new_aliases = existing + list(to_add)
        # QUOTE every value (existing too — re-emitting them quoted is harmless and
        # repairs any previously-unquoted entry):
        aliases_lines = '\n'.join(f'  - {_yaml_quote_alias(a)}' for a in new_aliases)
        new_block = f'aliases:\n{aliases_lines}'
        ALIASES_RE = re.compile(r'^aliases:.*?(?=^\w|\Z)', re.MULTILINE | re.DOTALL)
        new_fm = ALIASES_RE.sub(new_block + '\n', fm_section, count=1)
        if new_fm == fm_section:
            print(f"  warn: --fix could not locate aliases block in {os.path.relpath(fpath)} "
                  f"(skipped to avoid corruption)", file=sys.stderr)
            return  # Regex did not match; skip to avoid corruption
        new_content = new_fm + content[len(fm_section):]
        with open(fpath, 'w', encoding='utf-8') as f:
            f.write(new_content)
        add_finding('info', 'autofix', os.path.relpath(fpath),
                    f'Added self-aliases: {list(to_add)}')
    except Exception as e:
        print(f"  warn: --fix failed on {os.path.relpath(fpath)}: {e}", file=sys.stderr)
```
NOTE on idempotency with quoting: because emitted values are double-quoted, after a fix the parsed `aliases` are clean strings whose `.lower()` matches the title/id on the next run → `to_add` computes empty → no-op. Verified by Test 6 (idempotency) which now includes a colon-titled page.

# ---------------------------------------------------------------------------
# Shared normalization helper (D-02: used by linkres + reconciled orphan/gap)
# ---------------------------------------------------------------------------

_PLURAL_MAP = {
    'contexts': 'context',
    'policies': 'policy',
    'contracts': 'contract',
}
_PUNCT_RE = re.compile(r'[^\w\s]')   # strips parens, hyphens, punctuation characters
_WS_RE    = re.compile(r'\s+')

def normalize_link(text):
    """D-02: casefold, strip punctuation chars (parens stripped as CHARACTERS not
    content -- 'Hack (Agentive Stack)' -> 'hack agentive stack', NOT 'hack'),
    collapse whitespace, apply small explicit plural map (contexts->context,
    policies->policy, contracts->contract). NO edit distance, NO stemmer."""
    s = text.casefold()
    s = _PUNCT_RE.sub(' ', s)   # parens/hyphen/punct -> space
    s = s.replace('_', ' ')     # re \w includes underscore; handle separately
    s = _WS_RE.sub(' ', s).strip()
    words = [_PLURAL_MAP.get(w, w) for w in s.split()]
    return ' '.join(words)

def _yaml_quote_alias(a):
    """YAML-double-quote a scalar so colons/brackets/leading indicators stay strings.
    CONFIRMED HIGH BUG #2 fix: an unquoted '  - Topic: Subtitle' parses as a DICT."""
    s = str(a).replace('\\', '\\\\').replace('"', '\\"')
    return f'"{s}"'

def _apply_self_alias_fix(fpath, fm, to_add):
    """Idempotent: append the already-computed missing aliases (to_add: list of raw
    strings) to the page's aliases block. Caller computes to_add ONCE per page (title
    and/or id) and calls this ONCE — no stale-fm second call (review F / Codex). Never
    removes existing aliases. Regex confined to the frontmatter section (up to second
    '---') to avoid matching body 'aliases:' text (Pitfall 2). Emits YAML-double-quoted
    values so colon-bearing titles stay STRINGS (HIGH BUG #2)."""
    if not to_add:
        return  # nothing missing — idempotent no-op
    try:
        content = open(fpath, encoding='utf-8').read()
        fm_end = content.find('\n---', 3)  # skip opening ---
        fm_section = content[:fm_end] if fm_end != -1 else content
        existing = [str(a) for a in (fm.get('aliases') or []) if a]
        new_aliases = existing + list(to_add)
        # Quote EVERY value (existing too — harmless, and repairs any prior unquoted entry):
        aliases_lines = '\n'.join(f'  - {_yaml_quote_alias(a)}' for a in new_aliases)
        new_block = f'aliases:\n{aliases_lines}'
        ALIASES_RE = re.compile(r'^aliases:.*?(?=^\w|\Z)', re.MULTILINE | re.DOTALL)
        new_fm = ALIASES_RE.sub(new_block + '\n', fm_section, count=1)
        if new_fm == fm_section:
            print(f"  warn: --fix could not locate aliases block in {os.path.relpath(fpath)} "
                  f"(skipped to avoid corruption)", file=sys.stderr)
            return  # Regex did not match; skip to avoid corruption
        new_content = new_fm + content[len(fm_section):]
        with open(fpath, 'w', encoding='utf-8') as f:
            f.write(new_content)
        add_finding('info', 'autofix', os.path.relpath(fpath),
                    f'Added self-aliases: {list(to_add)}')
    except Exception as e:
        # Review F / Codex LOW: surface batch-fix failures instead of swallowing them.
        print(f"  warn: --fix failed on {os.path.relpath(fpath)}: {e}", file=sys.stderr)
