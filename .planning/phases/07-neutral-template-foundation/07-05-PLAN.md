    Follow the skeleton in Research Example 2, extended per behaviors:
    ```bash
    #!/usr/bin/env bash
    # bin/check-neutrality.sh -- NEUT-06 + NEUT-08 gate.
    # Greps public control-plane paths against a denylist (Kahneman + personal terms).
    # Exits 0 on clean; 1 on script failure; 2 on denylist match.
    set -euo pipefail

    ROOT="${PWD}"
    DENYLIST=".neutrality-denylist.txt"
    SUGGEST=0
    FORMAT="text"

    usage() { cat <<'EOF'
    Usage: bin/check-neutrality.sh [OPTIONS]
    Options:
      --root DIR           Scan root (default: PWD)
      --denylist FILE      Denylist file (default: .neutrality-denylist.txt under root)
      --format text|json   Output format (default: text)
      --suggest-denylist   Emit candidate terms from privacy: local_only pages; no denylist required
    Exit: 0 clean, 1 script failure, 2 denylist hit
    EOF
    }

    # flag parse ...

    PUBLIC_PATHS=(AGENTS.md CLAUDE.md README.md PRIVACY.md docs .github wiki bin)
    ```

    Core logic via a single python3 heredoc that:
      - In suggest mode: walks `$ROOT/wiki/` for YAML frontmatter with `privacy: local_only`; extracts distinctive tokens (slugs, proper nouns, multi-word phrases); filters out common English words (stopword list of ~50 terms: the, and, or, with, from, for, that, this, decision, personal, journal, ...); emits candidates, one per line, with a `# source: <path>` comment banner above each group.
      - In scan mode: loads denylist (strip `#` comments + blanks); for each term, `grep -rIinF` across `PUBLIC_PATHS` under `$ROOT`, skipping `examples/` via `--exclude-dir=examples`; collects `{path, line, term}` tuples; emits text (`PATH:LINE: term "..."`) or JSON based on `$FORMAT`.
      - Exits 2 if any hits found (scan mode).
      - Exits 1 if denylist file missing (scan mode).

    ```bash
    #!/usr/bin/env bash
    # bin/release.sh -- TMPL-11, D-04: orphan-branch publish to public remote.
    # Dry-run default; --apply requires interactive y/N; pre-flights check-neutrality.sh.
    set -euo pipefail

    APPLY=0
    REMOTE=""
    TAG="v1.1"

    usage() { cat <<'EOF'
    Usage: bin/release.sh --remote <url> [--tag <tag>] [--apply|--dry-run]
    Options:
      --remote URL   Target public remote (required)
      --tag TAG      Tag to create (default: v1.1)
      --dry-run      Default; print plan, no mutation
      --apply        Execute after y/N confirmation
    Exit: 0 ok/dry-run/abort, 1 missing args, 2 neutrality pre-flight failed
    EOF
    }

    while [ "$#" -gt 0 ]; do
        case "$1" in
            --help|-h) usage; exit 0 ;;
            --remote) REMOTE="$2"; shift 2 ;;
            --tag) TAG="$2"; shift 2 ;;
            --apply) APPLY=1; shift ;;
            --dry-run) APPLY=0; shift ;;
            *) echo "ERROR: unknown: $1" >&2; exit 1 ;;
        esac
    done

    [ -n "$REMOTE" ] || { echo "ERROR: --remote required" >&2; usage; exit 1; }

    WORKTREE="/tmp/gsd-release-${TAG}-$$"

    # Always compute + print plan
    echo "Target remote: $REMOTE"
    echo "Tag: $TAG"
    echo "Worktree: $WORKTREE"
    echo "Would include (public file set):"
    echo "  AGENTS.md CLAUDE.md README.md LICENSE PRIVACY.md"
    echo "  .gitignore .gitattributes .obsidianignore .neutrality-denylist.txt"
    echo "  bin/ schema/AGENTS.template.md"
    echo "  examples/kahneman/"
    echo "  wiki/index.md wiki/log.md wiki/decisions/"
    echo "  docs/"
    echo "  .github/workflows/neutrality.yml"
    echo "  .githooks/pre-commit"
    echo "Would exclude (belt-and-suspenders beyond .gitignore): .planning/ .brownfield/"

    if [ "$APPLY" -eq 0 ]; then
        echo "(dry-run) Pass --apply to execute."
        exit 0
    fi

    read -r -p "Proceed with publish? [y/N] " resp
    case "$resp" in
        y|Y|yes|YES) ;;
        *) echo "Aborted."; exit 0 ;;
    esac

    # --- Mutating path ---
    git worktree add --detach "$WORKTREE" HEAD
    trap 'git worktree remove --force "$WORKTREE" 2>/dev/null || true' EXIT

    cd "$WORKTREE"
    rm -rf .git
    git init --initial-branch=main >/dev/null

    # Belt-and-suspenders exclusion
    rm -rf .planning/ .brownfield/ .obsidian/workspace*.json

    # Pre-flight neutrality gate (uses the copied-over denylist + scripts)
    if ! bash bin/check-neutrality.sh --root . --denylist .neutrality-denylist.txt; then
        echo "NEUTRALITY PRE-FLIGHT FAILED in worktree. Aborting." >&2
        exit 2
    fi

    if ! bash bin/sync-claude.sh --check; then
        echo "CLAUDE.md drift in worktree. Aborting." >&2
        exit 2
    fi

    git add -A
    git -c user.email="release@<org>" -c user.name="release" commit -m "${TAG} release"
    git tag "$TAG"
    git push "$REMOTE" HEAD:main
    git push "$REMOTE" "$TAG"

    echo "Published ${TAG} to $REMOTE."
    ```
