    # dependency (ruamel.yaml is brownfield-path-only per CONTEXT.md §Established-Patterns).

    BF_REL_PATH="${DEST_FILE#$(pwd)/}"
    for BF_FIELD in bootstrap_stage bootstrap_date; do
        BF_VALUE="$(grep -E "^${BF_FIELD}:" "${DEST_FILE}" 2>/dev/null | head -1 | sed -E 's/^[^:]+:[[:space:]]*(.*)[[:space:]]*$/\1/')"
        if [ -n "${BF_VALUE}" ]; then

    import re, sys
    path, field = sys.argv[1], sys.argv[2]
    with open(path, 'r', encoding='utf-8') as f:
        content = f.read()
    # Strip the matching frontmatter line — anchored so body text containing
    # the token (e.g., inside a code block) is NOT touched. Requires the line
    # to start with `field:` and live inside the first `---` frontmatter block.
    # Heuristic: only strip if within the first 80 lines (frontmatter convention).
    lines = content.splitlines(keepends=True)
    in_fm = False; fm_end = 0
    for i, ln in enumerate(lines):
        if ln.strip() == '---':
            if not in_fm:
                in_fm = True
            else:
                fm_end = i; break
    new_lines = []
    for i, ln in enumerate(lines):
        if i <= fm_end and re.match(rf'^{re.escape(field)}\s*:', ln):
            continue  # drop this line
        new_lines.append(ln)
    with open(path, 'w', encoding='utf-8') as f:
        f.writelines(new_lines)
    PYSTRIP
            echo "Note: stripped ${BF_FIELD}=${BF_VALUE} from ${BF_REL_PATH} during ingest (brownfield-scoped field; see AGENTS.md §5)." >&2
