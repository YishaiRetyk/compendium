**Check 2: YAML frontmatter parses correctly.**
- Use inline python3 with PyYAML (per research recommendation — yq not available):
```bash
validate_frontmatter() {
    local file="$1"
    python3 -c "
import sys, yaml
content = open(sys.argv[1]).read()
if not content.startswith('---'):
    print('FAIL: No YAML frontmatter found', file=sys.stderr)
    sys.exit(1)
try:
    end = content.index('---', 3)
    fm = yaml.safe_load(content[3:end])
    if fm is None:
        print('FAIL: Empty frontmatter', file=sys.stderr)
        sys.exit(1)
    required = ['id','title','type','status','summary','created_at','updated_at',
                'sources','epistemic_status','tags','domains','privacy']
    missing = [f for f in required if f not in fm]
    if missing:
        print(f'FAIL: Missing required fields: {missing}', file=sys.stderr)
        sys.exit(1)
    print('PASS')
except yaml.YAMLError as e:
    print(f'FAIL: YAML parse error: {e}', file=sys.stderr)
    sys.exit(1)
except ValueError:
    print('FAIL: Unterminated frontmatter (missing closing ---)', file=sys.stderr)
    sys.exit(1)
" "$file"
}
```
Note: `supersedes`, `superseded_by`, and `aliases` are allowed to be null/empty, so only check the subset of fields that must always be present. The list above deliberately omits `supersedes`, `superseded_by`, and `aliases` because they are nullable.

**Check 3: Provenance references resolve to known source IDs in wiki/sources/.**
```bash
check_provenance() {
    local file="$1"
    local failed=0
    # Extract all [prov:source_id#...] markers
    local src_ids
    src_ids=$(grep -oP '\[prov:([^#\]]+)' "$file" 2>/dev/null | sed 's/\[prov://' | sort -u)
    if [ -z "$src_ids" ]; then
        echo "PASS (no provenance markers found)"
        return 0
    fi
    while IFS= read -r src_id; do
        if [ ! -f "wiki/sources/${src_id}.md" ]; then
            echo "FAIL: Provenance reference '${src_id}' does not resolve to wiki/sources/${src_id}.md" >&2
            failed=1
        fi
    done <<< "$src_ids"
    if [ "$failed" -eq 0 ]; then
        echo "PASS"
    fi
    return $failed
}
```

**Check 4: Privacy flags are respected.**
- Extract `privacy` field from frontmatter using python3 inline.
- For the target page, check that `privacy` is one of `local_only` or `cloud_safe`.
- For MERGE: if either source page has `privacy: local_only`, warn that the merged result must also be `local_only`.
```bash
check_privacy() {
    local file="$1"
    local privacy
    privacy=$(python3 -c "
import sys, yaml
content = open(sys.argv[1]).read()
end = content.index('---', 3)
fm = yaml.safe_load(content[3:end])
print(fm.get('privacy', 'MISSING'))
" "$file")
    if [ "$privacy" = "MISSING" ]; then
        echo "FAIL: No privacy field in frontmatter" >&2
        return 1
    fi
    if [ "$privacy" != "local_only" ] && [ "$privacy" != "cloud_safe" ]; then
        echo "FAIL: Invalid privacy value: $privacy (must be local_only or cloud_safe)" >&2
        return 1
    fi
    echo "$privacy"
    return 0
}
```
