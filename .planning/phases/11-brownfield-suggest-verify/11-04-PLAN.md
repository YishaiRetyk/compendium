**Item 4 — EOF-safe prompt loop (CRITICAL for CI/non-interactive safety):**

```python
_EOF_SENTINEL = object()
_MAX_REPROMPTS_PER_CLUSTER = 5

def prompt(msg):
    """Return user input (stripped) or _EOF_SENTINEL on stdin EOF.

    Review item 4 contract: distinguish zero-byte read (EOF) from '\\n' (Enter
    with no input). EOF ends the session cleanly; Enter-with-no-input is
    re-prompted, bounded by _MAX_REPROMPTS_PER_CLUSTER.
    """
    sys.stderr.write(msg)
    sys.stderr.flush()
    raw = sys.stdin.readline()
    if raw == '':
        return _EOF_SENTINEL   # true EOF (closed stdin)
    return raw.strip()

# In the per-cluster loop:
reprompts = 0
while True:
    choice = prompt("\n[a]pprove all / [r]eject all / [i]nspect / [o]verride / [s]kip: ")
    if choice is _EOF_SENTINEL:
        sys.stderr.write(f"\nreview-typing: stdin EOF detected; aborting session cleanly. "
                         f"{len(pending)} cluster(s) remain pending. Decisions written so far "
                         f"have been saved to {decisions_path}.\n")
        _eof_abort = True
        break   # breaks cluster loop, falls through to writeback
    c = (choice or '').strip().lower()[:1]
    if c in ('a', 'r', 'i', 'o', 's'):
        reprompts = 0
        # ... handle choice (a/r/i/o/s bodies unchanged from pre-review plan) ...
        if c in ('a', 'r', 's'):
            break   # cluster resolved or skipped
        continue   # i/o loop back to prompt
    else:
        reprompts += 1
        if reprompts >= _MAX_REPROMPTS_PER_CLUSTER:
            sys.stderr.write(f"review-typing: {_MAX_REPROMPTS_PER_CLUSTER} consecutive invalid inputs; "
                             f"skipping cluster {cid}.\n")
            break
        sys.stderr.write("Unrecognized choice. Try a/r/i/o/s.\n")
        continue
```

Break-to-writeback-and-exit flow: `_eof_abort = True` signals the outer for-loop to exit, and the writeback block runs so partial progress is persisted.

**Item 9 — stale candidate artifact detection in `verify` (read-only mode):**

```python
import hashlib, pathlib, re
def _body_post_op_hash_strip_sha256(path):
    body = pathlib.Path(path).read_bytes()
    lines = body.split(b'\n')
    stripped = [ln for ln in lines if not ln.startswith(b'# op_hash:') and not ln.startswith(b'# op_hash_scope:')]
    return 'sha256:' + hashlib.sha256(b'\n'.join(stripped)).hexdigest()

# For each canonical script that has a corresponding candidate YAML:
CANDIDATE_TO_SCRIPT = {
    'page-typing-candidates.yaml':    '01-page-typing.sh',
    'page-typing-decisions.yaml':     '01-page-typing.sh',
    'provenance-bootstrap-report.yaml': '02-provenance-bootstrap.sh',
    'cross-link-candidates.yaml':     '03-cross-link-inference.sh',
    'privacy-findings.yaml':          '04-privacy-review.sh',
}
HEADER_HASH_RE = re.compile(r'^# source_script_hash:\s*(sha256:[0-9a-f]+)\s*$', re.MULTILINE)

stale_warnings = []
for cand_name, script_name in CANDIDATE_TO_SCRIPT.items():
    cand_path = os.path.join(ROOT, '.brownfield', cand_name)
    script_path = os.path.join(ROOT, '.brownfield', 'migrations', script_name)
    if not (os.path.isfile(cand_path) and os.path.isfile(script_path)):
        continue   # candidate or script missing — out of scope for stale detection
    # Read header to get recorded hash
    with open(cand_path, 'r') as fh:
        head_text = ''.join(fh.readline() for _ in range(10))
    m = HEADER_HASH_RE.search(head_text)
    if not m:
        continue
    recorded = m.group(1)
    # Recompute the current script's body-post-op_hash-strip sha256
    current = _body_post_op_hash_strip_sha256(script_path)
    if recorded != current:
        stale_warnings.append((cand_name, script_name, recorded, current))

for cand, script, rec, cur in stale_warnings:
    sys.stderr.write(f"review-typing: stale candidate artifact detected: {cand} "
                     f"(source_script_hash in header {rec} != current {cur} for {script}); "
                     f"re-run `bin/brownfield.sh suggest` to refresh.\n")
# Exit 0 regardless — non-blocking WARN
```

    ```bash
            sf = False; bs = 0
            for i, ln in enumerate(cand_lines):
                if ln.strip() == '# ---':
                    if sf:
                        bs = i + 1; break
                    sf = True
            cand_data = yaml.load(''.join(cand_lines[bs:]))
            for c in cand_data.get('clusters', []):
                cand_by_id[c['cluster_id']] = list(c['pages'])
        for cluster_dec in dec.get('clusters', []):
            if cluster_dec.get('decision') == 'pending':
                for p in cand_by_id.get(cluster_dec['cluster_id'], []):
                    pending_pages.add(p)

    valid_types = set(VALID_ENUMS.get('type', ['entity', 'concept', 'source', 'comparison', 'overview', 'decision']))

    EXCLUDE_DIR_NAMES = {'.brownfield', '.git', '.obsidian', '.trash', 'node_modules'}
    promoted = []
    blocked = []
    for dirpath, dirnames, filenames in os.walk(ROOT, followlinks=False):
        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIR_NAMES]
        for fn in filenames:
            if not fn.endswith('.md'): continue
            full = os.path.join(dirpath, fn)
            rel = os.path.relpath(full, ROOT)
            try:
                fm, body, raw = read_fm_body(full)
            except Exception:
                continue
            if fm is None: continue

            # Gate 1
            if fm.get('bootstrap_stage') != 'bootstrapped':
                continue

            # Gate 2
            t = fm.get('type')
            if not t or t not in valid_types:
                blocked.append((rel, f'type not valid enum: {t!r}'))
                continue

            # Gate 3
            if rel in errors_by_path:
                blocked.append((rel, f'{len(errors_by_path[rel])} lint error(s)'))
                continue

            # Gate 4
            if t == 'source':
                required = ('path', 'content_hash', 'ingested_at', 'source_type')
                missing = [k for k in required if k not in fm or fm[k] in (None, '')]
                if missing:
                    blocked.append((rel, f'source missing required fields: {missing}'))
                    continue

            # Gate 5
            if rel in pending_pages:
                blocked.append((rel, 'page has pending review decision'))
                continue

            # All gates passed
            fm['bootstrap_stage'] = 'verified'
            write_roundtrip(full, fm, body, raw)
            promoted.append(rel)

    sys.stderr.write(f"verify --promote: {len(promoted)} page(s) promoted to verified; {len(blocked)} page(s) blocked\n")
    if promoted[:10]:
        for rel in promoted[:10]:
            sys.stderr.write(f"  [promoted] {rel}\n")
    if blocked[:10]:
        for rel, reason in blocked[:10]:
            sys.stderr.write(f"  [blocked]  {rel} — {reason}\n")
    PYEOF

        exit 0
    fi
    ```
