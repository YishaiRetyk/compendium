```python
def compute_op_hash(script_path: str, data_schema_version: int) -> str:
    body = pathlib.Path(script_path).read_bytes()
    # Strip ONLY op_hash header lines (canonical scripts ship without them per 11-01 Plan)
    lines = body.split(b'\n')
    stripped = [ln for ln in lines
                if not ln.startswith(b'# op_hash:')
                and not ln.startswith(b'# op_hash_scope:')]
    h = hashlib.sha256()
    h.update(b'\n'.join(stripped))
    h.update(f'\n# data_schema_version: {data_schema_version}\n'.encode())
    return f'sha256:{h.hexdigest()}'
```

```python
def cluster_is_autoapproveable(cluster: dict) -> bool:
    """Return True iff D-03's 'high confidence' gate is satisfied.

    D-03 verbatim: high confidence means '3+ signals agree OR explicit valid frontmatter type'.
    The cluster auto-approves (decisions: approve) in the initial decisions.yaml iff:
      - confidence == 'high', AND
      - EITHER signals.frontmatter is a valid type enum value (entity/concept/source/comparison/overview/decision)
              (the explicit-frontmatter path),
        OR 3 or more of the 5 non-frontmatter signals agree on the proposed_label
              (filename-shape + heading-shape + inbound-density + outbound-links + (future extensions) all vote).

    Counting 'signals agree' for the non-frontmatter path:
      - filename: 'kebab' agrees with concept/overview; 'pascal' agrees with entity; else 'none'.
      - heading:  'entity-like' agrees with entity; 'concept-like' agrees with concept; etc.
      - inbound:  'inbound-heavy' agrees with entity/overview; 'inbound-light' agrees with concept.
      - links:    'outbound-heavy' agrees with overview; 'outbound-light' agrees with entity; 'none' agrees with all.

    The 'agrees' dictionary is documented in the bin/lib/brownfield_classify.py docstring; this function uses
    the classifier's existing label-hint metadata. If in doubt, route to pending (conservative).
    """
    if cluster.get('confidence') != 'high':
        return False
    fm = (cluster.get('signals') or {}).get('frontmatter', 'none')
    VALID = {'entity', 'concept', 'source', 'comparison', 'overview', 'decision'}
    if fm in VALID:
        return True
    # Count non-frontmatter signals that agree with proposed_label
    label = cluster.get('proposed_label')
    signals = cluster.get('signals') or {}
    agreement_count = 0
    # Heuristic agreement tables (align with the classifier's label-hint mapping)
    LABEL_HINTS = {
        'entity':    {'filename': {'pascal'}, 'heading': {'entity-like'}, 'inbound': {'inbound-heavy'}, 'links': {'outbound-light', 'none'}},
        'concept':   {'filename': {'kebab'},  'heading': {'concept-like'}, 'inbound': {'inbound-light'}, 'links': {'outbound-heavy'}},
        'overview':  {'filename': {'kebab'},  'heading': {'overview-like'}, 'inbound': {'inbound-heavy'}, 'links': {'outbound-heavy'}},
        'source':    {'filename': {'date-prefixed'}, 'heading': {'source-like'}, 'inbound': {'inbound-light'}, 'links': {'none'}},
    }
    hints = LABEL_HINTS.get(label, {})
    for signal_name, accepted_values in hints.items():
        if signals.get(signal_name) in accepted_values:
            agreement_count += 1
    return agreement_count >= 3
```

    ```python
    # --- Phase 11 additions: cluster_by_signals() + cluster_is_autoapproveable() (BRWN-11, BRWN-14) ---

    from collections import defaultdict
    from typing import Iterable


    def cluster_by_signals(classifications: Iterable[dict]) -> list[dict]:
        """Group page classifications into clusters by signal tuple.

        Input
        -----
        classifications : iterable of dicts (output of classify_page() calls)
            Each dict has keys {path, label, confidence, signals, inbound_count}.

        Output
        ------
        list[dict] with keys:
            cluster_id      -- "cluster_1", "cluster_2", ... (1-indexed, stable
                               sort by signal tuple)
            page_count      -- int
            pages           -- sorted list of paths
            signals         -- {frontmatter, filename, heading, inbound, links}
            confidence      -- 'high' | 'medium' | 'low' | 'unknown'
            proposed_label  -- str (entity/concept/overview/source/comparison/decision/unknown)

        Rationale (RESEARCH Q1):
            - O(n) single pass; no scipy/numpy (zero new runtime deps).
            - Deterministic: sort() over the signal tuple; no hash() (PYTHONHASHSEED).
            - Human-reviewable: "cluster_1 (12 pages): PascalCase filenames,
              entity-like H1, inbound-heavy" beats an opaque similarity score.

        Invariant:
            All members of a cluster share the exact same signal tuple AND
            confidence band by construction. `proposed_label` is therefore
            identical across members — the function takes it from the first
            member.
        """
        # Materialize once (iterable may be single-pass)
        classes = list(classifications)
        buckets: dict[tuple, list[str]] = defaultdict(list)
        label_of: dict[tuple, str] = {}

        for c in classes:
            s = c.get('signals') or {}
            key = (
                s.get('frontmatter', 'none'),
                s.get('filename', 'none'),
                s.get('heading', 'none'),
                'inbound-heavy' if (c.get('inbound_count') or 0) >= 5 else 'inbound-light',
                s.get('links', 'none'),
                c.get('confidence', 'unknown'),
            )
            buckets[key].append(c['path'])
            if key not in label_of:
                label_of[key] = c.get('label', 'unknown')

        clusters = []
        # Sort on tuple of (all-string) keys -> fully deterministic across processes
        for idx, (key, pages) in enumerate(sorted(buckets.items()), start=1):
            clusters.append({
                'cluster_id': f'cluster_{idx}',
                'page_count': len(pages),
                'pages': sorted(pages),
                'signals': {
                    'frontmatter': key[0],
                    'filename': key[1],
                    'heading': key[2],
                    'inbound': key[3],
                    'links': key[4],
                },
                'confidence': key[5],
                'proposed_label': label_of[key],
            })
        return clusters


    # Label-hint agreement table — aligned with classify_page()'s signal derivation.
    # Each label lists the signal values that point TOWARD that label.
    # Used by cluster_is_autoapproveable() to count non-frontmatter signal agreement.
    _LABEL_HINTS = {
        'entity':    {'filename': {'pascal'},         'heading': {'entity-like'},   'inbound': {'inbound-heavy'}, 'links': {'outbound-light', 'none'}},
        'concept':   {'filename': {'kebab'},          'heading': {'concept-like'},  'inbound': {'inbound-light'}, 'links': {'outbound-heavy'}},
        'overview':  {'filename': {'kebab'},          'heading': {'overview-like'}, 'inbound': {'inbound-heavy'}, 'links': {'outbound-heavy'}},
        'source':    {'filename': {'date-prefixed'},  'heading': {'source-like'},   'inbound': {'inbound-light'}, 'links': {'none'}},
        'comparison':{'filename': {'kebab'},          'heading': {'comparison-like'},'inbound': {'inbound-light'},'links': {'outbound-heavy'}},
        'decision':  {'filename': {'dr-prefixed'},    'heading': {'decision-like'}, 'inbound': {'inbound-light'}, 'links': {'outbound-light'}},
    }
    _VALID_TYPE_ENUM = {'entity', 'concept', 'source', 'comparison', 'overview', 'decision'}


    def cluster_is_autoapproveable(cluster: dict) -> bool:
        """Return True iff D-03's 'high confidence' gate is satisfied for this cluster.

        D-03 verbatim (CONTEXT.md): high confidence means '3+ signals agree OR
        explicit valid frontmatter type'. Review item 7 widened this from the
        pre-review narrow frontmatter-only interpretation.

        Auto-approve iff ALL of:
          1. cluster['confidence'] == 'high'.
          2. EITHER signals.frontmatter is a valid type enum value (the explicit path),
             OR 3+ of the 5 non-frontmatter signals agree with the proposed_label
             via the _LABEL_HINTS table.

        Conservative default: return False if required keys are missing or
        proposed_label is 'unknown' (route to review queue).
        """
        if cluster.get('confidence') != 'high':
            return False
        signals = cluster.get('signals') or {}
        fm = signals.get('frontmatter', 'none')
        if fm in _VALID_TYPE_ENUM:
            return True   # explicit valid frontmatter type
        label = cluster.get('proposed_label')
        if not label or label == 'unknown':
            return False  # no target label to check agreement against
        hints = _LABEL_HINTS.get(label, {})
        if not hints:
            return False  # unknown label-hint mapping; conservative
        agreement_count = 0
        for signal_name, accepted_values in hints.items():
            if signals.get(signal_name) in accepted_values:
                agreement_count += 1
        return agreement_count >= 3
    ```

    ```python
    Consumed by:
      - bin/brownfield.sh scan branch (Phase 10 — refactor to import this)
      - bin/brownfield.sh suggest branch (Phase 11)

    """
    import os
    import re
    import fnmatch
    from typing import Iterator


    # Always-excluded directory basenames (hardcoded — matches Phase 10 behavior)
    HARDCODED_EXCLUDES = {'.brownfield', '.git', '.obsidian', '.trash', 'node_modules', '__pycache__'}


    def _load_ignore_patterns(root: str) -> list[str]:
        """Load .brownfield-ignore from root if present; return list of glob patterns.

        Format:
          - One pattern per line.
          - # comments ignored.
          - Blank lines ignored.
          - Trailing `/` means directory-only match.
          - Otherwise, gitignore-style glob match on path relative to root.
        """
        path = os.path.join(root, '.brownfield-ignore')
        if not os.path.isfile(path):
            return []
        patterns = []
        with open(path, 'r', encoding='utf-8') as fh:
            for raw in fh:
                s = raw.strip()
                if not s or s.startswith('#'):
                    continue
                patterns.append(s)
        return patterns


    def _matches_ignore(rel_path: str, patterns: list[str], is_dir: bool) -> bool:
        """Return True if rel_path matches any pattern. Mirrors Phase 10 semantics."""
        for p in patterns:
            if p.endswith('/'):
                if is_dir and (rel_path == p.rstrip('/') or rel_path.startswith(p)):
                    return True
                if (not is_dir) and rel_path.startswith(p):
                    return True
            else:
                if fnmatch.fnmatch(rel_path, p) or fnmatch.fnmatch(os.path.basename(rel_path), p):
                    return True
        return False


    def walk_vault_respecting_ignore(root: str, extra_ignores: list[str] | None = None) -> Iterator[str]:
        """Yield absolute paths of .md files under root, honoring exclusions.

        Exclusions applied (in order):
          1. Hardcoded excludes (HARDCODED_EXCLUDES).
          2. .brownfield-ignore patterns at root.
          3. extra_ignores (caller-provided; optional).
        """
        patterns = _load_ignore_patterns(root) + list(extra_ignores or [])
        root_abs = os.path.abspath(root)
        for dirpath, dirnames, filenames in os.walk(root_abs, followlinks=False):
            # Prune dirnames in-place to avoid descent into excluded trees
            pruned = []
            for d in dirnames:
                full_d = os.path.join(dirpath, d)
                rel_d = os.path.relpath(full_d, root_abs)
                if d in HARDCODED_EXCLUDES:
                    continue
                if _matches_ignore(rel_d, patterns, is_dir=True):
                    continue
                pruned.append(d)
            dirnames[:] = pruned
            for fn in filenames:
                if not fn.endswith('.md'):
                    continue
                full = os.path.join(dirpath, fn)
                rel = os.path.relpath(full, root_abs)
                if _matches_ignore(rel, patterns, is_dir=False):
                    continue
                yield full
    ```
