```python
def compute_op_hash(script_path: str, data_schema_version: int) -> str:
    body = pathlib.Path(script_path).read_bytes()
    # Strip ONLY op_hash header lines (canonical scripts ship without them per 11-01 Plan)
    lines = body.split(b'\n')
    stripped = [ln for ln in lines
                if not ln.startswith(b'# op_hash:')
                and not ln.startswith(b'# op_hash_scope:')]
    h = hashlib.sha256()
    h.update(b'\n'.join(stripped))
    h.update(f'\n# data_schema_version: {data_schema_version}\n'.encode())
    return f'sha256:{h.hexdigest()}'
```

    # --- Phase 11 addition: cluster_by_signals() (BRWN-11, BRWN-14) ---

    from collections import defaultdict
    from typing import Iterable


    def cluster_by_signals(classifications: Iterable[dict]) -> list[dict]:
        """Group page classifications into clusters by signal tuple.

        Input
        -----
        classifications : iterable of dicts (output of classify_page() calls)
            Each dict has keys {path, label, confidence, signals, inbound_count}.

        Output
        ------
        list[dict] with keys:
            cluster_id      -- "cluster_1", "cluster_2", ... (1-indexed, stable
                               sort by signal tuple)
            page_count      -- int
            pages           -- sorted list of paths
            signals         -- {frontmatter, filename, heading, inbound, links}
            confidence      -- 'high' | 'medium' | 'low' | 'unknown'
            proposed_label  -- str (entity/concept/overview/source/comparison/decision/unknown)

        Rationale (RESEARCH Q1):
            - O(n) single pass; no scipy/numpy (zero new runtime deps).
            - Deterministic: sort() over the signal tuple; no hash() (PYTHONHASHSEED).
            - Human-reviewable: "cluster_1 (12 pages): PascalCase filenames,
              entity-like H1, inbound-heavy" beats an opaque similarity score.

        Invariant:
            All members of a cluster share the exact same signal tuple AND
            confidence band by construction. `proposed_label` is therefore
            identical across members — the function takes it from the first
            member.
        """
        # Materialize once (iterable may be single-pass)
        classes = list(classifications)
        buckets: dict[tuple, list[str]] = defaultdict(list)
        label_of: dict[tuple, str] = {}

        for c in classes:
            s = c.get('signals') or {}
            key = (
                s.get('frontmatter', 'none'),
                s.get('filename', 'none'),
                s.get('heading', 'none'),
                'inbound-heavy' if (c.get('inbound_count') or 0) >= 5 else 'inbound-light',
                s.get('links', 'none'),
                c.get('confidence', 'unknown'),
            )
            buckets[key].append(c['path'])
            if key not in label_of:
                label_of[key] = c.get('label', 'unknown')

        clusters = []
        # Sort on tuple of (all-string) keys -> fully deterministic across processes
        for idx, (key, pages) in enumerate(sorted(buckets.items()), start=1):
            clusters.append({
                'cluster_id': f'cluster_{idx}',
                'page_count': len(pages),
                'pages': sorted(pages),
                'signals': {
                    'frontmatter': key[0],
                    'filename': key[1],
                    'heading': key[2],
                    'inbound': key[3],
                    'links': key[4],
                },
                'confidence': key[5],
                'proposed_label': label_of[key],
            })
        return clusters
    ```

        except Exception:
            return None, raw
        if not isinstance(fm, dict):
            return None, raw
        return fm, m.group(2)

    # Build inbound-link graph (one pass over vault)
    inbound = {}
    WIKILINK_RE = re.compile(r'\[\[([^\]|#]+)')
    pages_data = []
    for p in walk_vault(ROOT):
        fm, body = parse_fm_body(p)
        pages_data.append((p, fm, body))
        for m in WIKILINK_RE.finditer(body or ''):
            target_title = m.group(1).strip()
            inbound[target_title] = inbound.get(target_title, 0) + 1

    # Title-to-inbound lookup: resolve by title / filename stem
    def inbound_count_for(p, fm):
        title = (fm or {}).get('title') or pathlib.Path(p).stem
        return inbound.get(title, 0)

    # Classify each page
    classifications = []
    bootstrapped_paths = []
    for p, fm, body in pages_data:
        rel = os.path.relpath(p, ROOT)
        ic = inbound_count_for(p, fm)
        c = classify_page(rel, fm, body or '', inbound_count=ic)
        classifications.append(c)
        if fm and fm.get('bootstrap_stage') == 'bootstrapped':
            bootstrapped_paths.append(rel)

    # Cluster
    clusters = cluster_by_signals(classifications)

    # ===== Write page-typing-candidates.yaml =====
    def write_metadata_header(fh, source_script):
        fh.write('# ---\n')
        fh.write('# schema_version: 1\n')
        fh.write(f'# tool_version: {TOOL}\n')
        fh.write(f'# generated_at: {GENERATED_AT}\n')
        fh.write(f'# vault_root: {ROOT}\n')
        fh.write(f'# source_script_hash: {SCRIPT_HASHES[source_script]}\n')
        fh.write('# ---\n')

    candidates_path = os.path.join(BF_DIR, 'page-typing-candidates.yaml')
    with open(candidates_path, 'w', encoding='utf-8') as fh:
        write_metadata_header(fh, '01-page-typing.sh')
        pyyaml.safe_dump({'clusters': clusters}, fh, sort_keys=False, default_flow_style=False, allow_unicode=True)

    # ===== Write page-typing-decisions.yaml (initial: all pending except high-conf w/ explicit type:) =====
    decisions = {'clusters': []}
    for cl in clusters:
        # High-confidence auto-approve policy (D-03): high confidence ONLY if frontmatter signal shows explicit type
        auto_approve = (cl['confidence'] == 'high' and cl['signals']['frontmatter'] not in ('none', ''))
        decisions['clusters'].append({
            'cluster_id': cl['cluster_id'],
            'page_count': cl['page_count'],
            'proposed_label': cl['proposed_label'],
            'confidence': cl['confidence'],
            'decision': 'approve' if auto_approve else 'pending',
            'resolved_label': cl['proposed_label'] if auto_approve else None,
            'overrides': [],
        })

    decisions_path = os.path.join(BF_DIR, 'page-typing-decisions.yaml')
    with open(decisions_path, 'w', encoding='utf-8') as fh:
        write_metadata_header(fh, '01-page-typing.sh')
        pyyaml.safe_dump(decisions, fh, sort_keys=False, default_flow_style=False, allow_unicode=True)

    # ===== Write provenance-bootstrap-report.yaml (02 dry-run preview; eligible-count per page) =====
    # 02 scans TL;DR + Key Facts top-level bullets; Plan 11-02 emits a COUNT + sample.
    # Full eligibility logic is canonical in schema/brownfield/migrations/02-provenance-bootstrap.sh (Plan 11-03);
    # suggest produces a preview so users can gauge vault impact.
    WIKILINK_ONLY = re.compile(r'^\s*-\s*\[\[[^\]]+\]\]\s*$')
    QUESTION = re.compile(r'\?\s*$')
    TASK = re.compile(r'^\s*-\s*(\[[ x]\]|TODO:?|FIXME:?)\b', re.IGNORECASE)
    SOURCE_ID = re.compile(r'^\s*-\s*src-\d{4}-\d{2}-\d{2}-')
    PLACEHOLDER = re.compile(r'^\s*-\s*(TBD|TBC|pending|placeholder)\b', re.IGNORECASE)
    BULLET = re.compile(r'^(\s*)-\s+(.+)$')
    EP_PRESENT = re.compile(r'\[epistemic::')
    PV_PRESENT = re.compile(r'\[prov:')

    def is_eligible(line):
        if not BULLET.match(line): return False
        if WIKILINK_ONLY.match(line): return False
        if QUESTION.search(line): return False
        if TASK.match(line): return False
        if SOURCE_ID.match(line): return False
        if PLACEHOLDER.match(line): return False
        if EP_PRESENT.search(line): return False
        if PV_PRESENT.search(line): return False
        return True

    def section_scan(body, targets):
        eligible = []
        in_target = False
        HDR = re.compile(r'^##\s+(.+?)\s*$')
        for ln in body.splitlines():
            m = HDR.match(ln)
            if m:
                in_target = m.group(1).strip() in targets
                continue
            if in_target and is_eligible(ln):
                eligible.append(ln)
        return eligible

    prov_report = {'pages': []}
    for p, fm, body in pages_data:
        if not fm or fm.get('bootstrap_stage') != 'bootstrapped':
            continue
        rel = os.path.relpath(p, ROOT)
        eligible = section_scan(body or '', ['TL;DR', 'Key Facts'])
        entry = {
            'path': rel,
            'eligible_bullets': len(eligible),
            'sample': eligible[:3],
        }
        if not eligible:
            entry['note'] = 'no eligible claim bullets found'
        prov_report['pages'].append(entry)

    prov_path = os.path.join(BF_DIR, 'provenance-bootstrap-report.yaml')
    with open(prov_path, 'w', encoding='utf-8') as fh:
        write_metadata_header(fh, '02-provenance-bootstrap.sh')
        pyyaml.safe_dump(prov_report, fh, sort_keys=False, default_flow_style=False, allow_unicode=True)

    # ===== Write cross-link-candidates.yaml (03 output; match on titles with >=2 tokens OR >=8 chars) =====
    # Build title map from vault pages
    TITLE_MAP = {}  # title -> rel_path
    ALIAS_MAP = {}  # alias -> rel_path
    for p, fm, body in pages_data:
        rel = os.path.relpath(p, ROOT)
        t = (fm or {}).get('title') or pathlib.Path(p).stem
        TITLE_MAP[t] = rel
        for a in ((fm or {}).get('aliases') or []):
            ALIAS_MAP[str(a)] = rel

    CODE_FENCE = re.compile(r'^```')
    EXISTING_WL = re.compile(r'\[\[[^\]]+\]\]')
    def eligible_title(t):
        if not t: return False
        tokens = t.split()
        return len(tokens) >= 2 or len(t) >= 8

    cross_links = {'candidates': []}
    for p, fm, body in pages_data:
        if not body: continue
        rel = os.path.relpath(p, ROOT)
        in_fence = False
        seen_pairs = set()  # (source, target) — first-mention-only
        for ln_no, ln in enumerate(body.splitlines(), start=1):
            if CODE_FENCE.match(ln):
                in_fence = not in_fence
                continue
            if in_fence:
                continue
            # mask existing wikilinks
            masked = EXISTING_WL.sub(lambda m: ' ' * len(m.group(0)), ln)
            for title, target in list(TITLE_MAP.items()) + list(ALIAS_MAP.items()):
                if target == rel: continue   # don't link page to itself
                if not eligible_title(title): continue
                pattern = re.compile(r'\b' + re.escape(title) + r'\b')
                if pattern.search(masked):
                    pair = (rel, target)
                    if pair in seen_pairs: continue   # first-mention-only per AGENTS.md §8
                    seen_pairs.add(pair)
                    existing_link = EXISTING_WL.search(body) and ('[[' + title + ']]') in body
                    match_type = 'exact-title' if title in TITLE_MAP else 'alias'
                    cross_links['candidates'].append({
                        'source_page': rel,
                        'line_number': ln_no,
                        'matched_text': ln.strip()[:120],
                        'proposed_target': target,
                        'match_type': match_type,
                        'target_already_linked_from_source': bool(existing_link),
                    })
                    break   # move on to next line

    cross_links_path = os.path.join(BF_DIR, 'cross-link-candidates.yaml')
    with open(cross_links_path, 'w', encoding='utf-8') as fh:
        write_metadata_header(fh, '03-cross-link-inference.sh')
        pyyaml.safe_dump(cross_links, fh, sort_keys=False, default_flow_style=False, allow_unicode=True)

    # ===== Write privacy-findings.yaml (04 output; regex scan; NEVER flips privacy:) =====
    EMAIL_RE = re.compile(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b')
    PHONE_RE = re.compile(r'\b(?:\+?1[-.\s]?)?\(?[2-9][0-9]{2}\)?[-.\s]?[0-9]{3}[-.\s]?[0-9]{4}\b')
    SSN_RE = re.compile(r'\b\d{3}[-\s]\d{2}[-\s]\d{4}\b')
    EXAMPLE_TLDS = ('@example.com', '@example.org', '@example.net', '@test.invalid')
    BACKTICK = re.compile(r'`[^`]*`')

    privacy = {'findings': []}
    for p, fm, body in pages_data:
        if not body: continue
        rel = os.path.relpath(p, ROOT)
        in_fence = False
        for ln_no, ln in enumerate(body.splitlines(), start=1):
            if CODE_FENCE.match(ln):
                in_fence = not in_fence
                continue
            if in_fence:
                continue
            masked = BACKTICK.sub(lambda m: ' ' * len(m.group(0)), ln)
            for m in EMAIL_RE.finditer(masked):
                if any(tld in m.group(0) for tld in EXAMPLE_TLDS):
                    continue
                privacy['findings'].append({
                    'path': rel, 'line_number': ln_no,
                    'pattern_type': 'email', 'matched_text': m.group(0),
                    'risk_level': 'medium',
                    'suggestion': 'Review whether this email should appear in a cloud_safe page',
                })
            for m in PHONE_RE.finditer(masked):
                privacy['findings'].append({
                    'path': rel, 'line_number': ln_no,
                    'pattern_type': 'phone', 'matched_text': m.group(0),
                    'risk_level': 'medium',
                    'suggestion': 'Review whether this phone number should appear in a cloud_safe page',
                })
            for m in SSN_RE.finditer(masked):
                privacy['findings'].append({
                    'path': rel, 'line_number': ln_no,
                    'pattern_type': 'ssn', 'matched_text': '[redacted-SSN]',   # do not emit actual SSN into file
                    'risk_level': 'high',
                    'suggestion': 'SSN-like pattern detected; verify this is not real PII',
                })

    privacy_path = os.path.join(BF_DIR, 'privacy-findings.yaml')
    with open(privacy_path, 'w', encoding='utf-8') as fh:
        write_metadata_header(fh, '04-privacy-review.sh')
        pyyaml.safe_dump(privacy, fh, sort_keys=False, default_flow_style=False, allow_unicode=True)

    # ===== Extend REPORT.md with Phase 11 advisory sections (per Open-Q 4) =====
    report_path = os.path.join(BF_DIR, 'REPORT.md')
    with open(report_path, 'a', encoding='utf-8') as fh:
        fh.write('\n## Cross-link candidates\n\n')
        if cross_links['candidates']:
            for c in cross_links['candidates']:
                mark = '(already linked)' if c['target_already_linked_from_source'] else ''
                fh.write(f"- `{c['source_page']}`:{c['line_number']} -> `{c['proposed_target']}` ({c['match_type']}) {mark}\n")
        else:
            fh.write('_No cross-link candidates found._\n')
        fh.write('\n## Privacy review\n\n')
        if privacy['findings']:
            for f in privacy['findings']:
                fh.write(f"- `{f['path']}`:{f['line_number']} — {f['pattern_type']} ({f['risk_level']})\n")
        else:
            fh.write('_No privacy-sensitive patterns detected._\n')

    # ===== Summary to stderr =====
    sys.stderr.write(f"suggest: byte-copied 4 migration scripts to {BF_DIR}/migrations/\n")
    sys.stderr.write(f"suggest: wrote {len(clusters)} typing clusters ({sum(1 for c in decisions['clusters'] if c['decision'] == 'pending')} pending / {sum(1 for c in decisions['clusters'] if c['decision'] == 'approve')} auto-approved)\n")
    sys.stderr.write(f"suggest: wrote {len(prov_report['pages'])} provenance-bootstrap eligible previews\n")
    sys.stderr.write(f"suggest: wrote {len(cross_links['candidates'])} cross-link candidates\n")
    sys.stderr.write(f"suggest: wrote {len(privacy['findings'])} privacy findings\n")
    if not bootstrapped_paths:
        sys.stderr.write(f"suggest: WARN no bootstrapped pages found — run `bin/brownfield.sh bootstrap --apply` first for best results\n")
    PYEOF

        exit 0
    fi
    ```
